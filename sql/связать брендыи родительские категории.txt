WITH RECURSIVE subcategories AS (
    SELECT id, name, parent_id
    FROM category
    WHERE parent_id IS NOT NULL
    UNION ALL
    SELECT c.id, c.name, c.parent_id
    FROM category c
    INNER JOIN subcategories sc ON sc.id = c.parent_id
)

INSERT INTO category_brand (category_id, brand_id)
SELECT sc.parent_id, p.brand_id
FROM product p 
INNER JOIN subcategories sc ON sc.id = p.category_id
WHERE p.brand_id IS NOT NULL
ON CONFLICT DO NOTHING;


WITH RECURSIVE parent_categories AS (
    SELECT id, name 
    FROM category
    WHERE parent_id IS NULL

)
INSERT INTO category_brand (category_id, brand_id)
SELECT pc.id, p.brand_id
FROM product p 
INNER JOIN parent_categories pc ON pc.id = p.category_id
WHERE p.brand_id IS NOT NULL
ON CONFLICT DO NOTHING;
